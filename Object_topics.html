JavaScript Object Important Notes

1. Object Basics

const user = {
  name: "Manisha",
  age: 25,
};

console.log(user["name"]); // Manisha
console.log(user.name);    // Manisha

let key = "name";
console.log(user[key]);    // Manisha

user.name = "Shilpa";
console.log(user.name);    // Shilpa

user.gender = "female";
console.log(user.gender);  // female

delete user.gender;
console.log(user.gender);  // undefined

2. Dynamic Keys (Important)

user[key] = "Ankita";
console.log(user[key]); // Ankita

delete user[key];
console.log(user[key]); // undefined

3. Object Methods

const obj1 = { a: 3, b: 5, c: 9 };

Object.keys(obj1);    // ['a','b','c']
Object.values(obj1);  // [3,5,9]
Object.entries(obj1); // [['a',3], ['b',5], ['c',9]]

typeof obj1;                  // object
typeof Object.keys(obj1);     // object (array)

4. Shallow Copy

const obj2 = {
  name: "Manisha",
  address: { city: "Delhi" }
};

const shallowCopy = { ...obj2 };

shallowCopy.name = "Shilpa";                 // separate
shallowCopy.address.city = "Uttarakhand";    // SAME reference

console.log(obj2.address.city); // Uttarakhand

👉 Nested objects share reference

5. Deep Copy

const deepCopy = JSON.parse(JSON.stringify(obj2));

deepCopy.address.city = "Haldwani";

console.log(obj2.address.city); // NOT changed

👉 Completely independent copy

6. Object.assign()

const obj3 = { name: "Manisha" };
const obj4 = { gender: "Female" };

const merged = Object.assign(obj3, obj4);

console.log(merged); // { name: "Manisha", gender: "Female" }

7. this Keyword

const obj = {
  name: "Manisha",
  getName: function() {
    console.log(this.name);
  }
};

obj.getName(); // Manisha

let fn = obj.getName;
fn(); // undefined (lost context)

8. Object.freeze()

const frozenUser = { name: "Manisha" };

Object.freeze(frozenUser);

frozenUser.name = "Kanika"; // Not allowed
frozenUser.age = 25;        // Not allowed

console.log(frozenUser); // unchanged

👉 Cannot add, update, or delete

9. Object.seal()

const sealedUser = { name: "Chita" };

Object.seal(sealedUser);

sealedUser.name = "Kanika"; // allowed
sealedUser.age = 25;        // NOT allowed
delete sealedUser.name;     // NOT allowed

👉 Can update, but cannot add or delete

10. Prototype

const baseObj = { name: "Kanika" };

const child1 = Object.create(baseObj);
console.log(child1.name); // inherited from baseObj

const child2 = Object.create(child1);
console.log(child2.name); // chain works

11. hasOwnProperty

console.log(child2.hasOwnProperty("name"));  // false (inherited)
console.log(baseObj.hasOwnProperty("name")); // true (own property)

12. Closure

function createClosure() {
  let a = 10;

  function innerFunction() {
    console.log(a);
  }

  return innerFunction;
}

let closureTest = createClosure();
closureTest(); // 10

👉 Inner function remembers outer variable
